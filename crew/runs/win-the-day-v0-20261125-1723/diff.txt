+  [state on-toggle]

+      (when state
+        (let [{:keys [layout title collapse rows pill]} state]

+              (set! (.. box -style -width) (str (:width layout) "px"))
+              (set! (.. box -style -maxHeight) (str (:max-height layout) "px"))
